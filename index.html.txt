const STATUS_KEY = "hospital_status_v4";
const DELETED_KEY = "deleted_hospitals_v1";

/* =========================================================
   توابع کمکی
   ========================================================= */

function normalizeText(value){

    return String(value || "")
        .toLowerCase()
        .replace(/ي/g,"ی")
        .replace(/ى/g,"ی")
        .replace(/ك/g,"ک")
        .replace(/ة/g,"ه")
        .replace(/ۀ/g,"ه")
        .replace(/ؤ/g,"و")
        .replace(/إ/g,"ا")
        .replace(/أ/g,"ا")
        .replace(/ٱ/g,"ا")
        .replace(/‌/g," ")
        .replace(/\u200c/g," ")
        .replace(/\s+/g," ")
        .trim();

}


function toEnglishDigits(value){

    return String(value || "")
        .replace(/[۰-۹]/g,d =>
            String("۰۱۲۳۴۵۶۷۸۹".indexOf(d))
        )
        .replace(/[٠-٩]/g,d =>
            String("٠١٢٣٤٥٦٧٨٩".indexOf(d))
        );

}


function hospitalStatusKey(province,id){

    return normalizeText(province) + "_" + id;

}


function getHospitalStatus(province,id){

    const key = hospitalStatusKey(province,id);

    return localStorage.getItem(
        STATUS_KEY + "_" + key
    )
    ||
    initialHospitalStatuses[province]?.[id]
    ||
    "no-answer";

}


function setHospitalStatus(province,id,status){

    localStorage.setItem(
        STATUS_KEY + "_" +
        hospitalStatusKey(province,id),
        status
    );

}


function getDeletedHospitals(){

    try{

        return JSON.parse(
            localStorage.getItem(DELETED_KEY) || "{}"
        );

    }
    catch(e){

        return {};

    }

}


function isHospitalDeleted(province,id){

    return !!getDeletedHospitals()[
        hospitalStatusKey(province,id)
    ];

}


function getVisibleHospitals(province){

    return (hospitals[province] || [])
        .filter(h =>
            !isHospitalDeleted(province,h.id)
        );

}


/* =========================================================
   استان‌ها
   ========================================================= */

function renderProvinces(){

    const grid =
        document.getElementById("provinceGrid");

    const input =
        document.getElementById("provinceSearch");

    const clear =
        document.getElementById("provinceClear");

    const search =
        normalizeText(input.value);

    clear.classList.toggle(
        "show",
        !!search
    );

    const filtered =
        provinces.filter(p =>
            normalizeText(p).includes(search)
        );

    grid.innerHTML = "";

    if(!filtered.length){

        grid.innerHTML = `
            <div class="empty">
                استانی با این نام پیدا نشد.
            </div>
        `;

        return;

    }

    filtered.forEach(province => {

        const list =
            getVisibleHospitals(province);

        const counts = {

            ready:list.filter(
                h => getHospitalStatus(
                    province,h.id
                ) === "ready"
            ).length,

            noAnswer:list.filter(
                h => getHospitalStatus(
                    province,h.id
                ) === "no-answer"
            ).length,

            catalog:list.filter(
                h => getHospitalStatus(
                    province,h.id
                ) === "catalog"
            ).length,

            followUp:list.filter(
                h => getHospitalStatus(
                    province,h.id
                ) === "follow-up"
            ).length,

            emed:list.filter(
                h => getHospitalStatus(
                    province,h.id
                ) === "emed"
            ).length

        };


        const card =
            document.createElement("div");

        card.className =
            "province-card";

        card.onclick =
            () => openProvince(province);


        card.innerHTML = `

            <div class="province-name">
                ${escapeHTML(province)}
            </div>

            <div class="province-count">
                ${list.length} بیمارستان
            </div>

            ${
                list.length
                ?
                `
                <div class="province-status">

                    <span class="mini-status">
                        آماده: ${counts.ready}
                    </span>

                    <span class="mini-status">
                        بی‌پاسخ: ${counts.noAnswer}
                    </span>

                    <span class="mini-status">
                        کاتالوگ: ${counts.catalog}
                    </span>

                    <span class="mini-status">
                        پیگیری: ${counts.followUp}
                    </span>

                    ${
                        counts.emed

                        `
                        <span class="mini-status">
                            eMed: ${counts.emed}
                        </span>
                        `
                        :
                        ""
                    }

                </div>
                `
                :
                `
                <div
                    class="province-count"
                    style="margin-top:10px"
                >
                    بدون بیمارستان
                </div>
                `
            }

        `;

/* =========================================================
   جستجوی بیمارستان
   ========================================================= */

function hospitalMatches(
    hospital,
    province,
    search
){

    if(!search)
        return true;

    const tokens =
        normalizeText(search)
            .split(" ")
            .filter(Boolean);

    const status =
        getHospitalStatus(
            province,
            hospital.id
        );

    const searchable =
        normalizeText([

            hospital.name,
            hospital.city,
            statusNames[status] || "",
            hospital.note || "",
            ...(hospital.phones || [])

        ].join(" "));


    return tokens.every(
        token =>
            searchable.includes(token)
    );

}


/* =========================================================
   شماره تلفن
   ========================================================= */

function phoneHref(phone){

    return toEnglishDigits(phone)
        .replace(/[^\d+]/g,"");

}


/* =========================================================
   کلاس وضعیت
   ========================================================= */

function statusClass(status){

    if(status === "ready")
        return "status-ready";

    if(status === "catalog")
        return "status-catalog";

    if(status === "follow-up")
        return "status-follow-up";

    if(status === "emed")
        return "status-emed";

    return "status-no-answer";

}


/* =========================================================
   نمایش بیمارستان‌ها
   ========================================================= */

function renderHospitals(province){

    const grid =
        document.getElementById(
            "hospitalGrid"
        );

    const input =
        document.getElementById(
            "hospitalSearch"
        );

    const clear =
        document.getElementById(
            "hospitalClear"
        );

    const search = input.value;

    clear.classList.toggle(
        "show",
        !!normalizeText(search)
    );


    const list =
        getVisibleHospitals(province)
            .filter(h =>
                hospitalMatches(
                    h,
                    province,
                    search
                )
            );


    grid.innerHTML = "";


    if(!list.length){

        grid.innerHTML = `
            <div class="empty">
                بیمارستانی با این جستجو پیدا نشد.
            </div>
        `;

        return;

    }


    list.forEach(hospital => {

        const status =
            getHospitalStatus(
                province,
                hospital.id
            );


        const card =
            document.createElement("div");

        card.className =
            "hospital-card";

        card.onclick =
            () =>
                openHospital(
                    province,
                    hospital.id
                );


        const phonesHTML =
            (hospital.phones || [])
            .map(phone => `

                <a
                    class="card-phone"
                    href="tel:${phoneHref(phone)}"
                    onclick="event.stopPropagation()"
                >
                    📞 ${escapeHTML(phone)}
                </a>

            `)
            .join("");


        card.innerHTML = `

            <div class="hospital-title-row">

                <div class="hospital-name">
                    ${escapeHTML(hospital.name)}
                </div>

                <button
                    class="delete-btn"
                    title="حذف بیمارستان"
                    onclick="
                        deleteHospital(
                            event,
                            '${escapeJS(province)}',
                            ${hospital.id}
                        )
                    "
                >
                    🗑
                </button>

            </div>


            <div class="hospital-city">
                📍 ${escapeHTML(hospital.city || "")}
            </div>


            <div
                class="status-badge ${statusClass(status)}"
            >
                ${escapeHTML(
                    statusNames[status] ||
                    "تماس بی‌پاسخ"
                )}
            </div>


            <div>
                ${phonesHTML}
            </div>


            ${
                hospital.note
                ?
                `
                <div class="note">
                    📝 ${escapeHTML(
                        hospital.note
                    )}
                </div>
                `
                :
                ""
            }

        `;


        grid.appendChild(card);

    });

}


/* =========================================================
   باز کردن بیمارستان
   ========================================================= */

function openHospital(province,id){

    const hospital =
        (hospitals[province] || [])
        .find(h => h.id === id);


    if(
        !hospital ||
        isHospitalDeleted(
            province,
            id
        )
    )
        return;


    currentHospital =
        hospital;


    document.getElementById(
        "modalTitle"
    ).textContent =
        hospital.name;


    document.getElementById(
        "modalCity"
    ).textContent =
        "📍 " +
        (hospital.city || "");


    const currentStatus =
        getHospitalStatus(
            province,
            id
        );


    const statusButtons =
        document.getElementById(
            "statusButtons"
        );


    statusButtons.innerHTML = "";


    Object.entries(statusNames)
        .forEach(([key,label]) => {

            const btn =
                document.createElement(
                    "button"
                );

            btn.className =
                "status-btn";


            if(currentStatus === key)
                btn.classList.add("active");


            btn.textContent =
                label;


            btn.onclick = () => {

                setHospitalStatus(
                    province,
                    id,
                    key
                );

                closeModal();

                renderStats(province);

                renderHospitals(province);

                showToast(
                    "وضعیت به «" +
                    label +
                    "» تغییر کرد"
                );

            };


            statusButtons.appendChild(btn);

        });


    const phones =
        document.getElementById(
            "modalPhones"
        );


    phones.innerHTML = "";


    (hospital.phones || [])
        .forEach(phone => {

            const a =
                document.createElement(
                    "a"
                );

            a.className =
                "modal-phone";

            a.href =
                "tel:" +
                phoneHref(phone);

            a.textContent =
                "📞 " + phone;

            phones.appendChild(a);

        });


    const note =
        document.getElementById(
            "modalNote"
        );


    if(hospital.note){

        note.style.display =
            "block";

        note.textContent =
            "📝 " +
            hospital.note;

    }
    else{

        note.style.display =
            "none";

        note.textContent =
            "";

    }


    document.getElementById(
        "modal"
    ).classList.add("show");

}


/* =========================================================
   حذف بیمارستان
   ========================================================= */

function deleteHospital(
    event,
    province,
    id
){

    event.stopPropagation();


    const hospital =
        (hospitals[province] || [])
        .find(h => h.id === id);


    if(!hospital)
        return;


    if(
        !confirm(
            `آیا از حذف «${hospital.name}» مطمئن هستید؟\n\nاین بیمارستان از لیست سایت حذف می‌شود.`
        )
    )
        return;


    const deleted =
        getDeletedHospitals();


    deleted[
        hospitalStatusKey(
            province,
            id
        )
    ] = true;


    localStorage.setItem(
        DELETED_KEY,
        JSON.stringify(deleted)
    );


    showToast(
        "بیمارستان حذف شد"
    );


    if(currentProvince === province){

        renderStats(province);
        renderHospitals(province);

    }

}

function exportTXT(){

    let output = "";

    output +=
        "========================================\n";

    output +=
        "گزارش اطلاعات بیمارستان‌ها\n";

    output +=
        "========================================\n\n";

    output +=
        "تاریخ تهیه گزارش: " +
        new Date().toLocaleString("fa-IR") +
        "\n\n";


    let total = 0;


    provinces.forEach(
        province => {

            const list =
                getVisibleHospitals(
                    province
                );


            if(!list.length)
                return;


            output +=
                "\n########################################\n";

            output +=
                "استان: " +
                province +
                "\n";

            output +=
                "تعداد: " +
                list.length +
                "\n";

            output +=
                "########################################\n\n";


            list.forEach(
                (hospital,index) => {

                    total++;


                    const status =
                        getHospitalStatus(
                            province,
                            hospital.id
                        );


                    output +=
                        `${index+1}. ${hospital.name}\n`;

                    output +=
                        `شهر: ${hospital.city || ""}\n`;

                    output +=
                        `وضعیت: ${
                            statusNames[status] || ""
                        }\n`;


                    output +=
                        "تلفن‌ها:\n";


                    (hospital.phones || [])
                        .forEach(phone => {

                            output +=
                                "  " +
                                phone +
                                "\n";

                        });


                    if(hospital.note){

                        output +=
                            "یادداشت: " +
                            hospital.note +
                            "\n";

                    }


                    output +=
                        "----------------------------------------\n\n";

                }
            );

        }
    );


    output +=
        "\n========================================\n";

    output +=
        "تعداد کل بیمارستان‌های فعال: " +
        total +
        "\n";

    output +=
        "========================================\n";


    const blob =
        new Blob(
            ["\uFEFF" + output],
            {
                type:
                    "text/plain;charset=utf-8"
            }
        );


    const url =
        URL.createObjectURL(blob);


    const a =
        document.createElement("a");


    a.href = url;

    a.download =
        "hospital-data-" +
        new Date()
            .toISOString()
            .slice(0,10) +
        ".txt";


    document.body.appendChild(a);

    a.click();

    a.remove();

    URL.revokeObjectURL(url);


    showToast(
        "فایل اطلاعات ساخته شد"
    );

}

        document.getElementById(

        );

    );

                );

            },

        );

}


/* =========================================================
   امنیت HTML
   ========================================================= */

function escapeHTML(value){

    return String(value ?? "")
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");

}


function escapeJS(value){

    return String(value ?? "")
        .replace(/\\/g,"\\\\")
        .replace(/'/g,"\\'");

}
